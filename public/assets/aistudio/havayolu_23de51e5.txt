  <script>
    /**
     * Kilo ve Ventilasyon Hesaplayıcı Fonksiyonu
     * @param {Object} params - Hesaplama parametreleri
     * @param {string|number} params.height - Boy (cm veya inç)
     * @param {string|number} params.weight - Kilo (kg veya lb)
     * @param {boolean} [params.isMale=true] - Erkek ise true, Kadın ise false
     * @param {boolean} [params.isMetric=true] - Metrik (cm/kg) ise true, İmperial (inç/lb) ise false
     */
    function calculateWeightAndVentilation({ height, weight, isMale = true, isMetric = true }) {
      // 1. Birim Dönüşümleri (cm ve kg cinsine çevirme)
      const hCm = isMetric ? parseFloat(height) || 0 : (parseFloat(height) || 0) * 2.54;
      const wKg = isMetric ? parseFloat(weight) || 0 : (parseFloat(weight) || 0) * 0.45359237;

      // 2. İdeal Kilo (IBW - Devine Formülü)
      let ibw = 0;
      if (hCm > 152.4) {
        ibw = isMale ? 50.0 + 0.91 * (hCm - 152.4) : 45.5 + 0.91 * (hCm - 152.4);
      } else {
        ibw = isMale ? 50.0 : 45.5;
      }

      // 3. Vücut Kitle İndeksi (VKİ / BMI)
      const bmi = hCm > 0 ? wKg / Math.pow(hCm / 100.0, 2) : 0;

      // VKİ Sınıflandırması
      let bmiCategory = "";
      let bmiColor = "";
      if (bmi > 0) {
        if (bmi < 18.5) { bmiCategory = "Zayıf"; bmiColor = "#0288D1"; }
        else if (bmi < 25.0) { bmiCategory = "Normal"; bmiColor = "#2E7D32"; }
        else if (bmi < 30.0) { bmiCategory = "Fazla Kilolu"; bmiColor = "#EF6C00"; }
        else if (bmi < 35.0) { bmiCategory = "Obezite (Sınıf 1)"; bmiColor = "#C62828"; }
        else if (bmi < 40.0) { bmiCategory = "Obezite (Sınıf 2)"; bmiColor = "#B71C1C"; }
        else { bmiCategory = "Obezite (Sınıf 3 - Morbid)"; bmiColor = "#880E4F"; }
      }

      // 4. Öngörülen Kilo (PBW - Predicted Body Weight)
      const pbw = ibw; // ARDSnet standardı: PBW = IBW

      // 5. Düzeltilmiş Kilo (AdjBW - Adjusted Body Weight)
      // Gerçek kilo, İdeal kilonun %120'sinden fazlaysa hesaplanır
      const adjBw = (wKg > 0 && ibw > 0 && wKg > ibw * 1.2) 
        ? ibw + 0.4 * (wKg - ibw) 
        : wKg;

      // 6. Tidal Volüm Hesaplamaları (PBW üzerinden)
      const normalTvMin = pbw * 6; // Akciğer Koruyucu Ventilasyon (6 mL/kg)
      const normalTvMax = pbw * 8; // Akciğer Koruyucu Ventilasyon (8 mL/kg)
      const ardsTvMin = pbw * 4;   // ARDS Ventilasyonu (4 mL/kg)
      const ardsTvMax = pbw * 6;   // ARDS Ventilasyonu (6 mL/kg)

      return {
        hCm,
        wKg,
        ibw: ibw.toFixed(1),
        bmi: bmi.toFixed(1),
        bmiCategory,
        bmiColor,
        pbw: pbw.toFixed(1),
        adjBw: adjBw.toFixed(1),
        normalTv: `${Math.round(normalTvMin)} - ${Math.round(normalTvMax)} mL`,
        ardsTv: `${Math.round(ardsTvMin)} - ${Math.round(ardsTvMax)} mL`
      };
    }

    // DOM Elemanları
    const inputHeight = document.getElementById("inputHeight");
    const inputWeight = document.getElementById("inputWeight");
    const heightSuffix = document.getElementById("heightSuffix");
    const weightSuffix = document.getElementById("weightSuffix");

    const valBmi = document.getElementById("valBmi");
    const badgeBmi = document.getElementById("badgeBmi");
    const valIbw = document.getElementById("valIbw");
    const valAdjBw = document.getElementById("valAdjBw");
    const valNormalTv = document.getElementById("valNormalTv");
    const valArdsTv = document.getElementById("valArdsTv");

    function updateUI() {
      const isMetric = document.getElementById("unitMetric").checked;
      const isMale = document.getElementById("genderMale").checked;
      const height = parseFloat(inputHeight.value);
      const weight = parseFloat(inputWeight.value);

      // Sonekleri güncelle
      heightSuffix.textContent = isMetric ? "cm" : "in";
      weightSuffix.textContent = isMetric ? "kg" : "lb";

      // Eksik veya geçersiz girdi kontrolü
      if (!height || height <= 0) {
        valBmi.textContent = "-";
        valIbw.textContent = "-";
        valAdjBw.textContent = "-";
        valNormalTv.textContent = "-";
        valArdsTv.textContent = "-";
        badgeBmi.style.display = "none";
        return;
      }

      const results = calculateWeightAndVentilation({
        height,
        weight: weight || 0,
        isMale,
        isMetric
      });

      // IBW & Tidal Volümler sadece boya bağlıdır
      valIbw.textContent = `${results.ibw} kg`;
      valNormalTv.textContent = results.normalTv;
      valArdsTv.textContent = results.ardsTv;

      // Kilo gerektiren alanlar
      if (weight && weight > 0) {
        valBmi.textContent = `${results.bmi} kg/m²`;
        badgeBmi.textContent = results.bmiCategory;
        badgeBmi.style.backgroundColor = results.bmiColor;
        badgeBmi.style.display = "inline-block";
        valAdjBw.textContent = `${results.adjBw} kg`;
      } else {
        valBmi.textContent = "-";
        badgeBmi.style.display = "none";
        valAdjBw.textContent = "-";
      }
    }

    // Dinleyiciler (Event Listeners)
    inputHeight.addEventListener("input", updateUI);
    inputWeight.addEventListener("input", updateUI);

    document.querySelectorAll('input[name="unitSystem"]').forEach((elem) => {
      elem.addEventListener("change", (e) => {
        // İsteğe bağlı olarak mevcut değerleri dönüştür
        const toMetric = e.target.value === "metric";
        if (inputHeight.value) {
          inputHeight.value = toMetric
            ? (parseFloat(inputHeight.value) * 2.54).toFixed(1)
            : (parseFloat(inputHeight.value) / 2.54).toFixed(1);
        }
        if (inputWeight.value) {
          inputWeight.value = toMetric
            ? (parseFloat(inputWeight.value) * 0.45359237).toFixed(1)
            : (parseFloat(inputWeight.value) / 0.45359237).toFixed(1);
        }
        updateUI();
      });
    });

    document.querySelectorAll('input[name="gender"]').forEach((elem) => {
      elem.addEventListener("change", updateUI);
    });

    // Başlangıç tetiklemesi
    updateUI();
  </script>
